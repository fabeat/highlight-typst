<span class="hljs-comment">// Regression test: a content block `[...]` inside `parensMode`</span>
<span class="hljs-comment">// (i.e. as an argument to a function call, like</span>
<span class="hljs-comment">// `#box(text[Hello])`). Before the fix, `[` had no rule to</span>
<span class="hljs-comment">// match it inside `(...)`, so the markup text inside the</span>
<span class="hljs-comment">// content block was scanned by the code-mode variable rule</span>
<span class="hljs-comment">// and painted as `hljs-variable`. With the fix, a separate</span>
<span class="hljs-comment">// `nestedContentBlockMode` (no `endsParent: true`, so the</span>
<span class="hljs-comment">// closing `]` does not bubble up and break the enclosing</span>
<span class="hljs-comment">// parens) is wired into `parensMode.contains` and</span>
<span class="hljs-comment">// `bracesMode.contains` so `[...]` is recognised as a</span>
<span class="hljs-comment">// markup switch in any code-mode scope.</span>
<span class="hljs-comment">//</span>
<span class="hljs-comment">// Most visible in the user-reported `28. SEPTEMBER` line in</span>
<span class="hljs-comment">// the Spora plugin&#x27;s Datums-Pille example - the inner markup</span>
<span class="hljs-comment">// text `SEPTEMBER` was lighting up red before the fix.</span>

<span class="hljs-comment">// Simple: `[text]` inside `(...)` is markup, not code.</span>
#<span class="hljs-built_in">box</span>(<span class="hljs-built_in">text</span>[Hello])

<span class="hljs-comment">// With named arguments before the content block.</span>
#<span class="hljs-built_in">box</span>(<span class="hljs-built_in">text</span>(<span class="hljs-variable">size</span>: <span class="hljs-number">13pt</span>, <span class="hljs-variable">weight</span>: <span class="hljs-string">&quot;bold&quot;</span>)[Hello and stuff])

<span class="hljs-comment">// Multiple content-block arguments.</span>
#<span class="hljs-built_in">box</span>(<span class="hljs-built_in">pad</span>(<span class="hljs-variable">x</span>: <span class="hljs-number">50pt</span>, <span class="hljs-variable">y</span>: <span class="hljs-number">143pt</span>)[Hello], <span class="hljs-built_in">pad</span>(<span class="hljs-variable">x</span>: <span class="hljs-number">10pt</span>)[World])

<span class="hljs-comment">// Content block inside braces.</span>
#<span class="hljs-keyword">let</span> <span class="hljs-variable">f</span> = { <span class="hljs-variable">x</span>: [Hello] }

<span class="hljs-comment">// Multiple single-line content blocks inside a function call.</span>
#<span class="hljs-built_in">list</span>([a], [b], [c])
